// ================ Helpers ================
hlc3 = (high + low + close) / 3.0

// Rank current chart TF (higher rank = higher TF)
getRank() =>
    int r = 0
    if timeframe.ismonthly
        r := 10
    else if timeframe.isweekly
        r := 9
    else if timeframe.isdaily
        r := 8
    else
        s = timeframe.in_seconds()
        r := s >= 12*3600 ? 7 :
             s >=  6*3600 ? 6 :
             s >=  4*3600 ? 5 :
             s >=  1*3600 ? 4 :
             s >=   45*60 ? 3 :
             s >=   15*60 ? 2 :
             s >=    2*60 ? 1 :
             s >=       60 ? 0 :
                              -1 // 30s
    r

curRank = getRank()

// إظهار المجموعات التي رتبتها أكبر "strictly greater" من رتبة الفريم الحالي

show1   = iShow1m    and (curRank < 0)
